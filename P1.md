
using System;

class Program
{
    static void Main()
    {
        bool notifications = true;

        Console.WriteLine("1. فعال کردن اعلان‌ها");
        Console.WriteLine("2. غیرفعال کردن اعلان‌ها");
        Console.WriteLine("3. مشاهده وضعیت اعلان‌ها");
        Console.WriteLine("4. بازگرداندن تنظیمات پیش‌فرض");

        Console.Write("لطفاً گزینه موردنظر را وارد کنید: ");

        int choice;

        if (int.TryParse(Console.ReadLine(), out choice))
        {
            switch (choice)
            {
                case 1:
                    notifications = true;
                    Console.WriteLine("اعلان‌ها فعال شدند.");
                    break;

                case 2:
                    notifications = false;
                    Console.WriteLine("اعلان‌ها غیرفعال شدند.");
                    break;

                case 3:
                    if (notifications)
                    {
                        Console.WriteLine("وضعیت اعلان‌ها: فعال است.");
                    }
                    else
                    {
                        Console.WriteLine("وضعیت اعلان‌ها: غیرفعال است.");
                    }
                    break;

                case 4:
                    notifications = true;
                    Console.WriteLine("تنظیمات به حالت پیش‌فرض بازگشت.");
                    break;

                default:
                    Console.WriteLine("گزینه نامعتبر است.");
                    break;
            }
        }
        else
        {
            Console.WriteLine("گزینه نامعتبر است.");
        }
    }
}
