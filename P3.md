
using System;

class Program
{
    static void Main()
    {
        try
        {
            Console.Write("هزینه کل اشتراک را وارد کنید: ");
            double cost = Convert.ToDouble(Console.ReadLine());

            Console.Write("تعداد افراد را وارد کنید: ");
            int people = Convert.ToInt32(Console.ReadLine());

            if (cost < 0)
            {
                Console.WriteLine("هزینه نمی‌تواند منفی باشد.");
            }
            else if (people < 0)
            {
                Console.WriteLine("تعداد افراد نمی‌تواند منفی باشد.");
            }
            else if (people == 0)
            {
                Console.WriteLine("تعداد افراد نمی‌تواند صفر باشد.");
            }
            else
            {
                double share = cost / people;

                Console.WriteLine("سهم هر نفر: " + share + " تومان");
            }
        }
        catch (FormatException)
        {
            Console.WriteLine("ورودی نامعتبر است.");
        }
        catch (OverflowException)
        {
            Console.WriteLine("عدد واردشده خارج از محدوده مجاز است.");
        }
    }
}
